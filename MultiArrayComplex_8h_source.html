105template <size_t TWO_TIMES_RANK> requires ( !(TWO_TIMES_RANK%2) )
106Extents<TWO_TIMES_RANK/2> halveExtents(Extents<TWO_TIMES_RANK> extents)
107{
108 static constexpr size_t RANK=TWO_TIMES_RANK/2;
109 Extents<RANK> res;
110 std::ranges::for_each(std::views::iota(0ul,RANK), [&] (size_t i) {
111 res[i]=extents[i];
112#ifndef NDEBUG
113 if (size_t b=extents[i+RANK]; res[i]!=b)
114 throw std::invalid_argument("MultiArray does not have the shape of a matrix: index no. "+std::to_string(i)+": "+std::to_string(res[i])+" vs. "+std::to_string(b));
115#endif // NDEBUG
116 });
117 return res;
118}

143template <size_t RANK1, size_t RANK2>
144auto directProduct(const MultiArray<dcomp,RANK1>& m1, const MultiArray<dcomp,RANK2>& m2,
145 std::function<dcomp(dcomp,dcomp)> func=std::multiplies<dcomp>{} )
146{
147 MultiArray<dcomp,RANK1+RANK2> res{concatenate(m1.extents,m2.extents)};
148 for (size_t stride=m1.dataView.size(), i=0; i<stride; ++i) for (size_t j=0; j<m2.dataView.size(); ++j) res.dataStorage()[i+stride*j]=func(m1.dataView[i],m2.dataView[j]);
149 return res;
150}

175template <size_t TWO_TIMES_RANK>
176void hermitianConjugateSelf(MultiArrayView<dcomp,TWO_TIMES_RANK> mav)
177{
178 checkContiguous(mav,"hermitianConjugateSelf");
179 const size_t matrixDim = multiarray::calculateExtent(halveExtents(mav.extents));
180 auto _=[&] (size_t i, size_t j) -> dcomp& {return mav.dataView[i+matrixDim*j];};
181 for (size_t i=0; i<matrixDim; ++i) {
182 _(i,i)=conj(_(i,i));
183 for (size_t j=i+1; j<matrixDim; ++j) {
184 dcomp &u=_(i,j), &l=_(j,i);
185 u=conj(u); l=conj(l);
186 std::swap(u, l);
187 }
188 }
189}

199template <size_t TWO_TIMES_RANK>
200void twoTimesRealPartOfSelf(MultiArrayView<dcomp,TWO_TIMES_RANK> mav)
201{
202 checkContiguous(mav,"twoTimesRealPartOfSelf");
203 const size_t matrixDim = multiarray::calculateExtent(halveExtents(mav.extents));
204 auto _=[&] (size_t i, size_t j) -> dcomp& {return mav.dataView[i+matrixDim*j];};
205 for (size_t i=0; i<matrixDim; ++i) {
206 _(i,i)=2.*real(_(i,i));
207 for (size_t j=i+1; j<matrixDim; ++j) _(j,i)=conj(_(i,j)=_(i,j)+conj(_(j,i)));
208 }
209}

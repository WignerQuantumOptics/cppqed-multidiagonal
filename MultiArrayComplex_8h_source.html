95template <size_t TWO_TIMES_RANK> requires ( !(TWO_TIMES_RANK%2) )
96Extents<TWO_TIMES_RANK/2> halveExtents(Extents<TWO_TIMES_RANK> extents)
97{
98 static constexpr size_t RANK=TWO_TIMES_RANK/2;
99 Extents<RANK> res;
100 std::ranges::for_each(std::views::iota(0ul,RANK), [&] (size_t i) {
101 res[i]=extents[i];
102#ifndef NDEBUG
103 if (size_t b=extents[i+RANK]; res[i]!=b)
104 throw std::invalid_argument("MultiArray does not have the shape of a matrix: index no. "+std::to_string(i)+": "+std::to_string(res[i])+" vs. "+std::to_string(b));
105#endif // NDEBUG
106 });
107 return res;
108}

164template <size_t TWO_TIMES_RANK>
165void hermitianConjugateSelf(MultiArray<dcomp,TWO_TIMES_RANK>& ma)
166{
167 const size_t matrixDim = multiarray::calculateExtent(halveExtents(ma.extents));
168 for (size_t i=0; i<matrixDim; ++i) {
169 ma.dataStorage()[i+matrixDim*i]=conj(ma.dataStorage()[i+matrixDim*i]);
170 for (size_t j=i+1; j<matrixDim; ++j) {
171 dcomp
172 &u=ma.dataStorage()[i+matrixDim*j],
173 &l=ma.dataStorage()[j+matrixDim*i];
174 u=conj(u); l=conj(l);
175 std::swap(u, l);
176 }
177 }
178}

185template <size_t TWO_TIMES_RANK>
186void twoTimesRealPartOfSelf(MultiArrayView<dcomp,TWO_TIMES_RANK> mav)
187{
188 const size_t matrixDim = multiarray::calculateExtent(halveExtents(mav.extents));
189 auto _=[&] (size_t i, size_t j) -> dcomp& {return mav.dataView[i+matrixDim*j];};
190 for (size_t i=0; i<matrixDim; ++i) {
191 _(i,i)=2.*real(_(i,i));
192 for (size_t j=i+1; j<matrixDim; ++j) _(j,i)=conj(_(i,j)=_(i,j)+conj(_(j,i)));
193 }
194}

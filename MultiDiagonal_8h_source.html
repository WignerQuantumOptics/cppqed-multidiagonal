98 void operator () (double, MultiArrayConstView<dcomp,RANK> psi, MultiArrayView<dcomp,RANK> dpsidt) const
99 {
100 if (diagonals.empty()) return;
101
102#ifndef NDEBUG
103 if (psi.extents != calculateAndCheckDimensions(*this)) throw std::runtime_error("Mismatch between StateVector and MultiDiagonal dimensions");
104 checkExtents(psi,MultiArrayConstView<dcomp,RANK>{dpsidt},"MultiDiagonal application");
105#endif // NDEBUG
106
107 const size_t strideRows0{dpsidt.strides[0]}, strideCols0{psi.strides[0]};
108
109 for (const auto& [offsets,diag] : diagonals) {
110
111 if (diag.dataView.empty()) continue; // |o|=D on some axis: a structural zero
112
113 // first row o⁻ in dpsidt, first column o⁺ in psi
114 dcomp* rows{dpsidt.dataView.data()+dpsidt.offset}; const dcomp* cols{psi.dataView.data()+psi.offset};
115 for (auto&& [o,sR,sC] : std::views::zip(offsets,dpsidt.strides,psi.strides)) {
116 if (o<0) rows+=sR*size_t(-o); else cols+=sC*size_t(o);
117 }
118
119 // The loop nest walks the diagonal d in storage order (an owning MultiArray, contiguous, axis 0 fastest), and returns
120 // d advanced past the entries it used. len0 is the length of axis 0 if fixed at compile time, otherwise 0.
121 // Bounds and strides are copied, and d is passed by value, so that the compiler sees that the stores into dpsidt
122 // don't change them, also where it doesn't inline the lambda; otherwise it cannot vectorize.
123 const auto loop=[&] (auto n, const auto& self, auto len0, dcomp* r, const dcomp* c, const dcomp* d) -> const dcomp*
124 {
125 using namespace hana::literals;
126 if constexpr (n!=0_c) {
127 const size_t extent{diag.extents[n]}, sR{dpsidt.strides[n]}, sC{psi.strides[n]};
128 for (size_t k=0; k<extent; ++k) d=self(n-1_c,self,len0,r+k*sR,c+k*sC,d);
129 return d;
130 }
131 else {
132 const size_t sR{strideRows0}, sC{strideCols0};
133 const auto multiplyAdd=[=] (size_t k) {r[k*sR]+=textbookProduct(d[k],c[k*sC]);};
134 if constexpr (decltype(len0)::value) {
135 hana::for_each(hana::make_range(hana::size_c<0>,len0),multiplyAdd);
136 return d+decltype(len0)::value;
137 }
138 else {
139 const size_t len{diag.extents[0]};
140 if (sR==1 && sC==1) for (size_t k=0; k<len; ++k) r[k]+=textbookProduct(d[k],c[k]);
141 else for (size_t k=0; k<len; ++k) multiplyAdd(k);
142 return d+len;
143 }
144 }
145 };
146
147 const auto run=[&] (auto len0) {loop(hana::llong_c<RANK-1>,loop,len0,rows,cols,diag.dataView.data());};
148
149 // A short axis 0 is unrolled at compile time: a vectorized loop over a few elements costs more than it saves.
150 // Up to four is where the two cross over (benchmarks/kernels.cc, case E).
151 switch (diag.extents[0]) {
152 case 1: run(hana::size_c<1>); break;
153 case 2: run(hana::size_c<2>); break;
154 case 3: run(hana::size_c<3>); break;
155 case 4: run(hana::size_c<4>); break;
156 default: run(hana::size_c<0>);
157 }
158 }
159 }

179 MultiDiagonal& hermitianConjugate()
180 {
181 Diagonals newDiagonals;
182 for (auto& [offsets,diag] : diagonals) {
183 Offsets newOffsets;
184 for (size_t i=0; i<RANK; ++i) newOffsets[i]=-offsets[i];
185 conj(diag);
186 newDiagonals.emplace(newOffsets,std::move(diag));
187 }
188 diagonals.swap(newDiagonals);
189 return *this;
190 }

214 MultiDiagonal& operator+=(const MultiDiagonal& md)
215 {
216#ifndef NDEBUG
217 if (auto dim{calculateAndCheckDimensions(*this)}, mdDim{calculateAndCheckDimensions(md)};
218 dim!=Dimensions{} && mdDim!=Dimensions{} && dim!=mdDim)
219 throw std::runtime_error("Dimension mismatch in addition of MultiDiagonals");
220#endif // NDEBUG
221 for (const auto& [offsets,diag] : md.diagonals)
222 if (auto it=diagonals.find(offsets); it!=diagonals.end())
223 for (auto&& [to,from] : std::views::zip(it->second.dataStorage(),diag.dataStorage())) to+=from;
224 else diagonals.emplace(offsets,copy(diag));
225 return *this;
226 }

249 friend Dimensions calculateAndCheckDimensions(const MultiDiagonal& md)
250 {
251 if (md.diagonals.empty()) return {};
252
253 auto impliedDimensions = [] (const auto& kv) -> Dimensions {
254 auto res{kv.second.extents};
255 for (auto&& [v,o] : std::views::zip(res,kv.first)) v += size_t(o<0 ? -o : o);
256 return res;
257 };
258
259 const auto res{impliedDimensions(*md.diagonals.cbegin())};
260
261 for (const auto& kv : md.diagonals)
262 if (impliedDimensions(kv)!=res) throw std::runtime_error("Dimensions mismatch in MultiDiagonal");
263
264 return res;
265 }

297template <size_t RANK>
298MultiDiagonal<RANK> operator|(const MultiDiagonal<RANK>& a, const MultiDiagonal<RANK>& b)
299{
300 using MD = MultiDiagonal<RANK>;
301
302 if (a.diagonals.empty() || b.diagonals.empty()) return MD{}; // composition with the zero operator
303
304 const auto dim = calculateAndCheckDimensions(a);
305 if (dim != calculateAndCheckDimensions(b)) throw std::runtime_error("Mismatch in MultiDiagonal composition dimensions");
306
307 MD res;
308
309 for (const auto& [p,diagA] : a.diagonals) for (const auto& [q,diagB] : b.diagonals) {
310
311 typename MD::Offsets s; // s = p + q
312 Extents<RANK> resExtents, start, box; // result-diagonal extents; valid window [start, start+box)
313 std::array<ptrdiff_t,RANK> iA, iB; // index shifts into the operand diagonals
314
315 bool nonzero=true;
316
317 for (size_t i=0; i<RANK; ++i) {
318 const ptrdiff_t D=ptrdiff_t(dim[i]), pi=p[i], qi=q[i], si=pi+qi,
319 sm = si<0 ? -si : 0, pm = pi<0 ? -pi : 0, qm = qi<0 ? -qi : 0, // negative parts o⁻
320 sa = si<0 ? -si : si, pa = pi<0 ? -pi : pi, qa = qi<0 ? -qi : qi; // magnitudes |o|
321
322 if (sa >= D) {nonzero=false; break;}
323
324 const ptrdiff_t ia = sm - pm, ib = sm + pi - qm,
325 st = std::max({ptrdiff_t{0}, -ia, -ib}),
326 en = std::min({D - sa, D - pa - ia, D - qa - ib});
327
328 if (en <= st) {nonzero=false; break;}
329
330 s[i]=si; resExtents[i]=size_t(D-sa); start[i]=size_t(st); box[i]=size_t(en-st); iA[i]=ia; iB[i]=ib;
331 }
332 if (!nonzero) continue;
333
334 // find-or-create the result diagonal (zero-initialized); accumulate the windowed products
335 auto& resDiag = res.diagonals.try_emplace(s, resExtents, multiarray::zeroInit<dcomp>).first->second;
336
337 for (Extents<RANK> t{}; t[0]!=box[0]; incrementMultiIndex(t,box)) {
338 Extents<RANK> kRes, kA, kB;
339 for (size_t i=0; i<RANK; ++i) {
340 kRes[i]=start[i]+t[i];
341 kA[i]=size_t(ptrdiff_t(kRes[i])+iA[i]);
342 kB[i]=size_t(ptrdiff_t(kRes[i])+iB[i]);
343 }
344 resDiag(kRes) += diagA(kA)*diagB(kB);
345 }
346 }
347
348 return res;
349}

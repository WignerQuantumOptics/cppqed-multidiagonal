89 if (diagonals.empty()) return;
90
91#ifndef NDEBUG
92 if (psi.extents != calculateAndCheckDimensions(*this)) throw std::runtime_error("Mismatch between StateVector and MultiDiagonal dimensions");
93#endif // NDEBUG
94
95 for (const auto& [offsets,diag] : diagonals) {
96
97 if (diag.dataView.empty()) continue; // |o|=D on some axis: a structural zero
98
99 Dimensions ubound{psi.extents}, lboundLHS{}, lboundRHS{};
100 for (auto&& [o,u,lL,lR] : std::views::zip(offsets,ubound,lboundLHS,lboundRHS)) {
101 lL = o<0 ? size_t(-o) : 0uz; lR = o>0 ? size_t(o) : 0uz; u -= lR;
102 }

104 const auto increment=[&] (auto n, const auto& inc, Dimensions& idxLHS, Dimensions& idxRHS, Dimensions& idxDiag)
105 {
106 using namespace hana::literals;
107 if constexpr (n!=0_c) {
108 if (idxLHS[n]==ubound[n]-1) {idxLHS[n]=lboundLHS[n]; idxRHS[n]=lboundRHS[n]; idxDiag[n]=0; inc(n-1_c,inc,idxLHS,idxRHS,idxDiag);}
109 else {idxLHS[n]++; idxRHS[n]++; idxDiag[n]++;}
110 }
111 else {idxLHS[0]++; idxRHS[0]++; idxDiag[0]++;}

114 for ( Dimensions idxLHS{lboundLHS}, idxRHS{lboundRHS}, idxDiag{}; idxLHS[0]!=ubound[0] ; increment( hana::llong_c<RANK-1>, increment, idxLHS, idxRHS, idxDiag ) )
115 dpsidt(idxLHS) += diag(idxDiag)*psi(idxRHS);
116 }

137 MultiDiagonal& hermitianConjugate()
138 {
139 Diagonals newDiagonals;

141 Offsets newOffsets;
142 for (size_t i=0; i<RANK; ++i) newOffsets[i]=-offsets[i];

207 friend Dimensions calculateAndCheckDimensions(const MultiDiagonal& md)
208 {
209 if (md.diagonals.empty()) return {};
210
211 auto impliedDimensions = [] (const auto& kv) -> Dimensions {
212 auto res{kv.second.extents};
213 for (auto&& [v,o] : std::views::zip(res,kv.first)) v += size_t(o<0 ? -o : o);
214 return res;
215 };
216
217 const auto res{impliedDimensions(*md.diagonals.cbegin())};
218
219 for (const auto& kv : md.diagonals)
220 if (impliedDimensions(kv)!=res) throw std::runtime_error("Dimensions mismatch in MultiDiagonal");
221
222 return res;
223 }

255template <size_t RANK>
256MultiDiagonal<RANK> operator|(const MultiDiagonal<RANK>& a, const MultiDiagonal<RANK>& b)
257{
258 using MD = MultiDiagonal<RANK>;
259
260 if (a.diagonals.empty() || b.diagonals.empty()) return MD{}; // composition with the zero operator
261
262 const auto dim = calculateAndCheckDimensions(a);
263 if (dim != calculateAndCheckDimensions(b)) throw std::runtime_error("Mismatch in MultiDiagonal composition dimensions");
264
265 MD res;
266
267 for (const auto& [p,diagA] : a.diagonals) for (const auto& [q,diagB] : b.diagonals) {
268
269 typename MD::Offsets s; // s = p + q
270 Extents<RANK> resExtents, start, box; // result-diagonal extents; valid window [start, start+box)
271 std::array<ptrdiff_t,RANK> iA, iB; // index shifts into the operand diagonals
272
273 bool nonzero=true;
274
275 for (size_t i=0; i<RANK; ++i) {
276 const ptrdiff_t D=ptrdiff_t(dim[i]), pi=p[i], qi=q[i], si=pi+qi,
277 sm = si<0 ? -si : 0, pm = pi<0 ? -pi : 0, qm = qi<0 ? -qi : 0, // negative parts o⁻
278 sa = si<0 ? -si : si, pa = pi<0 ? -pi : pi, qa = qi<0 ? -qi : qi; // magnitudes |o|
279
280 if (sa >= D) {nonzero=false; break;}
281
282 const ptrdiff_t ia = sm - pm, ib = sm + pi - qm,
283 st = std::max({ptrdiff_t{0}, -ia, -ib}),
284 en = std::min({D - sa, D - pa - ia, D - qa - ib});
285
286 if (en <= st) {nonzero=false; break;}
287
288 s[i]=si; resExtents[i]=size_t(D-sa); start[i]=size_t(st); box[i]=size_t(en-st); iA[i]=ia; iB[i]=ib;
289 }
290 if (!nonzero) continue;
291
292 // find-or-create the result diagonal (zero-initialized); accumulate the windowed products
293 auto& resDiag = res.diagonals.try_emplace(s, resExtents, multiarray::zeroInit<dcomp>).first->second;
294
295 for (Extents<RANK> t{}; t[0]!=box[0]; incrementMultiIndex(t,box)) {
296 Extents<RANK> kRes, kA, kB;
297 for (size_t i=0; i<RANK; ++i) {
298 kRes[i]=start[i]+t[i];
299 kA[i]=size_t(ptrdiff_t(kRes[i])+iA[i]);
300 kB[i]=size_t(ptrdiff_t(kRes[i])+iB[i]);
301 }
302 resDiag(kRes) += diagA(kA)*diagB(kB);
303 }
304 }
305
306 return res;
307}

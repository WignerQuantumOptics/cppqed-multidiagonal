125 void applyByDiagonals(MultiArrayConstView<dcomp,RANK> psi, MultiArrayView<dcomp,RANK> dpsidt) const
126 {
127 if (!checkApplication(psi,dpsidt)) return;
128
129 const size_t strideRows0{dpsidt.strides[0]}, strideCols0{psi.strides[0]};
130
131 for (const auto& [offsets,diag] : diagonals) {
132
133 if (diag.dataView.empty()) continue; // |o|=D on some axis: a structural zero
134
135 // first row o⁻ in dpsidt, first column o⁺ in psi
136 dcomp* rows{dpsidt.dataView.data()+dpsidt.offset}; const dcomp* cols{psi.dataView.data()+psi.offset};
137 for (auto&& [o,sR,sC] : std::views::zip(offsets,dpsidt.strides,psi.strides)) {
138 if (o<0) rows+=sR*size_t(-o); else cols+=sC*size_t(o);
139 }
140
141 // The loop nest walks the diagonal d in storage order (an owning MultiArray, contiguous, axis 0 fastest), and returns
142 // d advanced past the entries it used. len0 is the length of axis 0 if fixed at compile time, otherwise 0.
143 // Bounds and strides are copied, and d is passed by value, so that the compiler sees that the stores into dpsidt
144 // don't change them, also where it doesn't inline the lambda; otherwise it cannot vectorize.
145 const auto loop=[&] (auto n, const auto& self, auto len0, dcomp* r, const dcomp* c, const dcomp* d) -> const dcomp*
146 {
147 using namespace hana::literals;
148 if constexpr (n!=0_c) {
149 const size_t extent{diag.extents[n]}, sR{dpsidt.strides[n]}, sC{psi.strides[n]};
150 for (size_t k=0; k<extent; ++k) d=self(n-1_c,self,len0,r+k*sR,c+k*sC,d);
151 return d;
152 }
153 else {
154 const size_t sR{strideRows0}, sC{strideCols0};
155 const auto multiplyAdd=[=] (size_t k) {r[k*sR]+=textbookProduct(d[k],c[k*sC]);};
156 if constexpr (decltype(len0)::value) {
157 hana::for_each(hana::make_range(hana::size_c<0>,len0),multiplyAdd);
158 return d+decltype(len0)::value;
159 }
160 else {
161 const size_t len{diag.extents[0]};
162 if (sR==1 && sC==1) for (size_t k=0; k<len; ++k) r[k]+=textbookProduct(d[k],c[k]);
163 else for (size_t k=0; k<len; ++k) multiplyAdd(k);
164 return d+len;
165 }
166 }
167 };
168
169 const auto run=[&] (auto len0) {loop(hana::llong_c<RANK-1>,loop,len0,rows,cols,diag.dataView.data());};
170
171 // A short axis 0 is unrolled at compile time: a vectorized loop over a few elements costs more than it saves.
172 // Up to four is where the two cross over (benchmarks/kernels.cc, case E).
173 switch (diag.extents[0]) {
174 case 1: run(hana::size_c<1>); break;
175 case 2: run(hana::size_c<2>); break;
176 case 3: run(hana::size_c<3>); break;
177 case 4: run(hana::size_c<4>); break;
178 default: run(hana::size_c<0>);
179 }
180 }
181 }

197 void applyByRows(MultiArrayConstView<dcomp,RANK> psi, MultiArrayView<dcomp,RANK> dpsidt) const
198 {
199 if (!checkApplication(psi,dpsidt)) return;
200
201 struct Tap {Offsets o; Extents<RANK> strides;}; // a diagonal: its offsets, and its strides in storage
202 std::array<Tap,maxListed> taps;
203
204 using Pointers = std::array<const dcomp*,maxListed>;
205 // the diagonals with entries in a row: their indices in taps, and where their entries and their columns start
206 struct Listed {size_t size; std::array<size_t,maxListed> tap; Pointers d, c;};
207
208 const auto loop=[&] (auto n, const auto& self, const Listed& listed, dcomp* r) -> void
209 {
210 using namespace hana::literals;
211 const size_t extent{dpsidt.extents[n]}, sR{dpsidt.strides[n]}, sC{psi.strides[n]};
212 const auto bounds=[&] (size_t i) {const ptrdiff_t o{taps[listed.tap[i]].o[n]}; return std::pair{size_t(o<0 ? -o : 0),size_t(o>0 ? o : 0)};};
213
214 if constexpr (n!=0_c) for (size_t k=0; k<extent; ++k) { // row k along axis n: entry k-o⁻ of each diagonal with k ∈ [o⁻,D-o⁺)
215 Listed next{};
216 for (size_t i=0; i<listed.size; ++i)
217 if (const auto [om,op]=bounds(i); k>=om && k<extent-op) {
218 const size_t t{listed.tap[i]};
219 next.tap[next.size]=t; next.d[next.size]=listed.d[i]+(k-om)*taps[t].strides[n]; next.c[next.size]=listed.c[i]+(k-om+op)*sC;
220 ++next.size;
221 }
222 if (next.size) self(n-1_c,self,next,r+k*sR);
223 }
224 else {
225 // rows [lo,hi) have entries in every listed diagonal
226 size_t lo=0, hi=extent;
227 for (size_t i=0; i<listed.size; ++i) {const auto [om,op]=bounds(i); lo=std::max(lo,om); hi=std::min(hi,extent-op);}
228
229 const auto sumOneByOne=[&] (size_t row) { // entry row-o⁻, column row-o⁻+o⁺
230 dcomp sum{};
231 for (size_t i=0; i<listed.size; ++i)
232 if (const auto [om,op]=bounds(i); row>=om && row<extent-op) sum+=textbookProduct(listed.d[i][row-om],listed.c[i][(row-om+op)*sC]);
233 r[row*sR]+=sum;
234 };
235 for (size_t row=0; row<lo; ++row) sumOneByOne(row);
236 for (size_t row=std::max(lo,hi); row<extent; ++row) sumOneByOne(row);
237
238 if (lo<hi) for (size_t i=0; i<listed.size; i+=maxSummed) {
239 const size_t count{std::min(maxSummed,listed.size-i)};
240 std::array<const dcomp*,maxSummed> d{}, c{};
241 for (size_t j=0; j<count; ++j) {const auto [om,op]=bounds(i+j); d[j]=listed.d[i+j]+(lo-om); c[j]=listed.c[i+j]+(lo-om+op)*sC;}
242 [&] <size_t... m> (std::index_sequence<m...>) {
243 ((count==m+1 ? sumRows(d,c,r+lo*sR,hi-lo,sR,sC,std::make_index_sequence<m+1>{}) : void()), ...);
244 } (std::make_index_sequence<maxSummed>{});
245 }
246 }
247 };
248
249 for (auto diag=diagonals.begin(); diag!=diagonals.end(); ) {
250 Listed all{};
251 for (; diag!=diagonals.end() && all.size<maxListed; ++diag) if (!diag->second.dataView.empty()) {
252 taps[all.size]={diag->first,diag->second.strides};
253 all.tap[all.size]=all.size; all.d[all.size]=diag->second.dataView.data(); all.c[all.size]=psi.dataView.data()+psi.offset;
254 ++all.size;
255 }
256 if (all.size) loop(hana::llong_c<RANK-1>,loop,all,dpsidt.dataView.data()+dpsidt.offset);
257 }
258 }

264 bool checkApplication(MultiArrayConstView<dcomp,RANK> psi, MultiArrayView<dcomp,RANK> dpsidt) const
265 {
266 if (diagonals.empty()) return false;
267#ifndef NDEBUG
268 if (psi.extents != calculateAndCheckDimensions(*this)) throw std::runtime_error("Mismatch between StateVector and MultiDiagonal dimensions");
269 checkExtents(psi,MultiArrayConstView<dcomp,RANK>{dpsidt},"MultiDiagonal application");
270#else
271 (void)psi; (void)dpsidt;
272#endif // NDEBUG
273 return true;
274 }

278 template <size_t... g>
279 static void sumRows(const std::array<const dcomp*,maxSummed>& dAll, const std::array<const dcomp*,maxSummed>& cAll,
280 dcomp* r, size_t len, size_t sR, size_t sC, std::index_sequence<g...>)
281 {
282 const std::array d{dAll[g]...}, c{cAll[g]...};
283 if (sR==1 && sC==1) for (size_t k=0; k<len; ++k) r[k]+=(... + textbookProduct(d[g][k],c[g][k]));
284 else for (size_t k=0; k<len; ++k) r[k*sR]+=(... + textbookProduct(d[g][k],c[g][k*sC]));
285 }

307 MultiDiagonal& hermitianConjugate()
308 {

311 Offsets newOffsets;
312 for (size_t i=0; i<RANK; ++i) newOffsets[i]=-offsets[i];

377 friend Dimensions calculateAndCheckDimensions(const MultiDiagonal& md)
378 {
379 if (md.diagonals.empty()) return {};
380
381 auto impliedDimensions = [] (const auto& kv) -> Dimensions {
382 auto res{kv.second.extents};
383 for (auto&& [v,o] : std::views::zip(res,kv.first)) v += size_t(o<0 ? -o : o);
384 return res;
385 };
386
387 const auto res{impliedDimensions(*md.diagonals.cbegin())};
388
389 for (const auto& kv : md.diagonals)
390 if (impliedDimensions(kv)!=res) throw std::runtime_error("Dimensions mismatch in MultiDiagonal");
391
392 return res;
393 }

425template <size_t RANK>
426MultiDiagonal<RANK> operator|(const MultiDiagonal<RANK>& a, const MultiDiagonal<RANK>& b)
427{
428 using MD = MultiDiagonal<RANK>;
429
430 if (a.diagonals.empty() || b.diagonals.empty()) return MD{}; // composition with the zero operator
431
432 const auto dim = calculateAndCheckDimensions(a);
433 if (dim != calculateAndCheckDimensions(b)) throw std::runtime_error("Mismatch in MultiDiagonal composition dimensions");
434
435 MD res;
436
437 for (const auto& [p,diagA] : a.diagonals) for (const auto& [q,diagB] : b.diagonals) {
438
439 typename MD::Offsets s; // s = p + q
440 Extents<RANK> resExtents, start, box; // result-diagonal extents; valid window [start, start+box)
441 std::array<ptrdiff_t,RANK> iA, iB; // index shifts into the operand diagonals
442
443 bool nonzero=true;
444
445 for (size_t i=0; i<RANK; ++i) {
446 const ptrdiff_t D=ptrdiff_t(dim[i]), pi=p[i], qi=q[i], si=pi+qi,
447 sm = si<0 ? -si : 0, pm = pi<0 ? -pi : 0, qm = qi<0 ? -qi : 0, // negative parts o⁻
448 sa = si<0 ? -si : si, pa = pi<0 ? -pi : pi, qa = qi<0 ? -qi : qi; // magnitudes |o|
449
450 if (sa >= D) {nonzero=false; break;}
451
452 const ptrdiff_t ia = sm - pm, ib = sm + pi - qm,
453 st = std::max({ptrdiff_t{0}, -ia, -ib}),
454 en = std::min({D - sa, D - pa - ia, D - qa - ib});
455
456 if (en <= st) {nonzero=false; break;}
457
458 s[i]=si; resExtents[i]=size_t(D-sa); start[i]=size_t(st); box[i]=size_t(en-st); iA[i]=ia; iB[i]=ib;
459 }
460 if (!nonzero) continue;
461
462 // find-or-create the result diagonal (zero-initialized); accumulate the windowed products
463 auto& resDiag = res.diagonals.try_emplace(s, resExtents, multiarray::zeroInit<dcomp>).first->second;
464
465 for (Extents<RANK> t{}; t[0]!=box[0]; incrementMultiIndex(t,box)) {
466 Extents<RANK> kRes, kA, kB;
467 for (size_t i=0; i<RANK; ++i) {
468 kRes[i]=start[i]+t[i];
469 kA[i]=size_t(ptrdiff_t(kRes[i])+iA[i]);
470 kB[i]=size_t(ptrdiff_t(kRes[i])+iB[i]);
471 }
472 resDiag(kRes) += diagA(kA)*diagB(kB);
473 }
474 }
475
476 return res;
477}

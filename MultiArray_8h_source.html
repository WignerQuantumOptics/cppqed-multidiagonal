60template <size_t RANK>
61auto& incrementMultiIndex(Extents<RANK>& idx, Extents<RANK> extents)
62{
63 const auto increment=[&](auto n, auto inc)
64 {
65 using namespace hana::literals;
66 if constexpr (n!=0_c) {
67 if (idx[n]==extents[n]-1) {idx[n]=0; inc(n-1_c,inc);}
68 else idx[n]++;
69 }
70 else idx[0]++; // This will eventually put the index into an illegal state, but this is how all (unchecked) iterators work.
71 };
72
73 increment(hana::llong_c<RANK-1>,increment);
74
75 return idx;
76}

104template <typename T, size_t RANK>
105class MultiArrayView
106{
107public:
108 MultiArrayView(const MultiArrayView&) = default; MultiArrayView(MultiArrayView&&) = default; MultiArrayView() = default;
109 MultiArrayView& operator=(const MultiArrayView&) = default; MultiArrayView& operator=(MultiArrayView&&) = default;
110
112 MultiArrayView(Extents<RANK> extents, Extents<RANK> strides, size_t offset, auto&&... dataView)
113 : extents{extents}, strides{strides}, offset{offset}, dataView{FWD(dataView)...} {}
114
116 operator MultiArrayView<const T, RANK>() const requires ( !std::is_const_v<T> ) {return {extents,strides,offset,dataView}; }
117
119 void checkBounds(std::convertible_to<size_t> auto ... i) const
120 {
121#ifndef NDEBUG
122 auto e=extents.begin();
123 (... , [&] {
124 if (auto eComp=e++; static_cast<size_t>(i) >= *eComp)
125 throw std::range_error("Index position: "+std::to_string(eComp-extents.begin())+", index value: "+std::to_string(i)+", extent: "+std::to_string(*eComp));
126 } () );
127#else
128 (... , (void)i); // silence unused warning
129#endif // NDEBUG
130 }
131
133
137 T& operator() (Extents<RANK> idx) const
138 {
139 std::apply([this] (auto ... args) { this->checkBounds(args ...); },idx);
140
141 return dataView[std::ranges::fold_left( std::views::zip(idx,strides), offset,
142 [&] (auto init, auto ids) {return init+get<0>(ids)*get<1>(ids);} ) ];
143 }
144
146 T& operator() (std::convertible_to<size_t> auto ... i) const requires (sizeof...(i)==RANK)
147 {
148 checkBounds(i...);
149 size_t idx=0;
150 auto s=strides.begin();
151 return dataView[ offset + ( ... , [&] { return idx+=(*s++)*i;} () ) ];
152 }
153
155 T& operator() (std::convertible_to<size_t> auto i) const requires (RANK==1) {checkBounds(i); return dataView[offset+strides[0]*i];}
156
158 T& operator() (std::convertible_to<size_t> auto i, std::convertible_to<size_t> auto j) const requires (RANK==2) {
159 checkBounds(i,j); return dataView[offset+strides[0]*i+strides[1]*j];
160 }
161
163 auto operator() (std::convertible_to<size_t> auto i, multiarray::Placeholder) const requires (RANK==2) {
164 return MultiArrayView<T,1>{ {extents[1]}, {strides[1]}, offset+i*strides[0], dataView};
165 }
166
168 auto operator() (multiarray::Placeholder, std::convertible_to<size_t> auto j) const requires (RANK==2) {
169 return MultiArrayView<T,1>{ {extents[0]}, {strides[0]}, offset+j*strides[1], dataView};
170 }
171
172 Extents<RANK> extents, strides;
173
174 size_t offset;
175
176 std::span<T> dataView;
177
178};

212template <typename T1, typename T2, size_t RANK>
213bool isEqual(MultiArrayConstView<T1,RANK> m1, MultiArrayConstView<T2,RANK> m2)
214{
215 checkExtents(m1,m2,"MultiArrayView comparison");
216 bool res=true;
217 for (Extents<RANK> idx{}; idx[0]!=m1.extents[0]; incrementMultiIndex(idx,m1.extents))
218 res &= ( m1(idx)==m2(idx) );
219 return res;
220}

230template <size_t RANK>
231auto calculateStrides(Extents<RANK> extents)
232{
233 Extents<RANK> strides; strides[0]=1;
234 for (size_t i=1; i<RANK; ++i) strides[i]=strides[i-1]*extents[i-1];
235 return strides;
236}

239template <size_t RANK>
240auto calculateExtent(Extents<RANK> extents)
241{
242 return std::ranges::fold_left(extents,1uz,std::multiplies{});
243}
